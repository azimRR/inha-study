Solution: Exceptional Control Flow (ECF)
A signal is a kind of (usually
software) interrupt, used to
announce asynchronous
events to a process.
There is a limited list of
possible signals; we do not
invent our own. (There might
be 64 signals, for instance.)
The name of a LINUX signal
begins with "SIG".
An asynchronous event is
a task or operation that is
initiated, but does not
block the main program
thread from continuing to
execute other code while
the task runs in the
background.
SIGINT is a signal
generated when a
user presses
Control-C.
This will terminate
the program from
the terminal.

// multi_signal.c
#include <stdio.h>
#include <stdlib.h>
#include <signal.h>
#include <unistd.h>
void handler(int sig) {
switch(sig) {
case SIGINT:
printf("\nCaught SIGINT (Ctrl+C)\n");
break;
case SIGTERM:
printf("\nCaught SIGTERM (termination request)\n");
break;
case SIGUSR1:
printf("\nCaught SIGUSR1 (user-defined 1)\n");
break;
}
}
int main() {
printf("Process %d waiting for signals...\n", getpid());
printf("Try: Ctrl+C (SIGINT) or kill -TERM %d (SIGTERM)\n\n", getpid());
signal(SIGINT, handler);
signal(SIGTERM, handler);
signal(SIGUSR1, handler);
while(1) {
pause();
// Wait for signals
}
return 0;
}

Signal Handling Examples
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <signal.h>
#include <errno.h>
#include <string.h>
// Error handling function
void unix_error(char *msg) {
fprintf(stderr, "%s: %s\n", msg,
strerror(errno));
exit(1);
}
// Handler for SIGINT (Ctrl+C)
void handler1(int sig) {
if (sig == SIGINT)
printf("Caught SIGINT (Ctrl+C)\n");
}
// Handler for SIGQUIT (Ctrl+\)
void handler2(int sig) {
if (sig == SIGQUIT)
printf("Caught SIGQUIT (Ctrl+\\)\n");
}
int main() {
// Install the SIGINT handler
printf("Installing SIGINT handler...\n");
if (signal(SIGINT, handler1) == SIG_ERR) {
unix_error("signal error for SIGINT");
}
// Install the SIGQUIT handler
printf("Installing SIGQUIT handler...\n");
if (signal(SIGQUIT, handler2) == SIG_ERR) {
unix_error("signal error for SIGQUIT");
}
printf("MAIN PROCESS NOW WAITING FOR signals...\n");
printf(" Ctrl+C (SIGINT) or Ctrl+\\ (SIGQUIT)\n");
printf("Ctrl+Z to suspend or 'kill -9 PID' \n\n");
// Wait for signals indefinitely
while (1) {
pause();
// Suspends until a signal arrives
}
exit(0);
}
