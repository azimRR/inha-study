Example: the writer/client
9/22/2026
SOC 3010              j.yusupov@inha.uz
FIFOs & pipelines
/* writer.c — creates the FIFO and sends one message */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>
#define FIFO_PATH "/tmp/demo_fifo"
int main(void){
    if (mkfifo(FIFO_PATH, 0666) == -1 && errno != EEXIST) {
        perror("mkfifo");
        exit(EXIT_FAILURE);
    }
    printf("writer: waiting for a reader (open blocks)...\n");
    int fd = open(FIFO_PATH, O_WRONLY);      /* < blocks here */
    if (fd == -1) { perror("open"); exit(EXIT_FAILURE); }
    printf("writer: reader arrived, sending\n");
    char *msg = "hello over FIFO\n";
    if (write(fd, msg, strlen(msg)) == -1) perror("write");
    close(fd);
    return 0;
}
The perror() function is a built-in C standard library function used to print
a human-readable error message to the standard error stream (stderr).

Example: the reader/server
9/22/2026
SOC 3010              j.yusupov@inha.uz
FIFOs & pipelines
/* reader.c — receives everything until the writer closes */
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#define FIFO_PATH "/tmp/demo_fifo"
int main(void)
{
    printf("reader: waiting for a writer (open blocks)...\n");
    int fd = open(FIFO_PATH, O_RDONLY);      /* < blocks here */
    if (fd == -1) { perror("open"); exit(EXIT_FAILURE); }
    printf("reader: writer arrived\n");
    char buf[256];
    ssize_t n;
    while ((n = read(fd, buf, sizeof buf)) > 0)
        write(STDOUT_FILENO, buf, n);
     /*n==0>all writers closed (EOF)*/
    printf("reader: EOF, done\n");
    close(fd);
    //unlink(PATH); to remove the FIFO
    return 0;
}
Pure shell, no C at all
